cmake_minimum_required(VERSION 3.16)
project(fpai_rf_worker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_CROSSCOMPILING)
  set(CMAKE_SKIP_RPATH TRUE)
  set(FPAI_TARGET_DEPENDENCY_ROOT "" CACHE PATH "Extracted ARM64 dependency root")
  if(FPAI_TARGET_DEPENDENCY_ROOT)
    foreach(fpai_libdir IN ITEMS usr/lib/aarch64-linux-gnu lib/aarch64-linux-gnu)
      link_directories("${FPAI_TARGET_DEPENDENCY_ROOT}/${fpai_libdir}")
      add_link_options("-Wl,-rpath-link,${FPAI_TARGET_DEPENDENCY_ROOT}/${fpai_libdir}")
    endforeach()
  endif()
endif()
if(MSVC)
  add_compile_options(/Zc:__cplusplus /utf-8)
endif()
find_package(Icraft-HostBackend 3.36.1.0 EXACT REQUIRED)
find_package(icraft-zg330backend 3.36.1.0 EXACT REQUIRED)
find_package(Threads REQUIRED)
add_executable(rf_worker src/rf_worker.cpp)
target_include_directories(rf_worker PRIVATE include)
target_link_libraries(rf_worker PRIVATE Icraft::HostBackend Icraft::ZG330Backend Threads::Threads)
if(WIN32)
  target_link_libraries(rf_worker PRIVATE ws2_32)
endif()
